import cadquery as cq
from math import isclose

# ============================================================
# Coordinate frame
# +X = Right
# +Y = Forward
# +Z = Up
#
# Origin (0, 0, 0):
# Centre of the bottom face of the base plate.
# ============================================================

# ----------------------------
# Parameters
# ----------------------------
base_len_x = 90.0
base_len_y = 50.0
base_thickness = 10.0

plate_corner_radius = 5.0

corner_hole_dia = 6.0
corner_hole_radius = corner_hole_dia / 2.0

boss_dia = 20.0
boss_radius = boss_dia / 2.0
boss_height = 50.0

central_hole_dia = 15.0
central_hole_radius = central_hole_dia / 2.0

cutout_y_size = 20.0   # 20 mm side parallel to short edge of plate
cutout_x_depth = 10.0  # inward from right short edge

final_fillet_spec = 1.0
final_fillet_radius = final_fillet_spec - 0.001

tol = 1.0e-6

# ----------------------------
# Derived global coordinates
# ----------------------------
x_min = -base_len_x / 2.0
x_max = +base_len_x / 2.0
y_min = -base_len_y / 2.0
y_max = +base_len_y / 2.0

z_base_bottom = 0.0
z_base_top = base_thickness
z_boss_top = z_base_top + boss_height

# Corner radius centres
corner_hole_centres = [
    (x_min + plate_corner_radius, y_min + plate_corner_radius),  # lower-left
    (x_max - plate_corner_radius, y_min + plate_corner_radius),  # lower-right
    (x_max - plate_corner_radius, y_max - plate_corner_radius),  # upper-right
    (x_min + plate_corner_radius, y_max - plate_corner_radius),  # upper-left
]

# Boss / central hole centre
boss_cx = x_min + 30.0
boss_cy = y_min + 20.0

# Right-edge rectangular cutout bounds
cutout_x0 = x_max - cutout_x_depth
cutout_x1 = x_max
cutout_y0 = -cutout_y_size / 2.0
cutout_y1 = +cutout_y_size / 2.0


# ============================================================
# Helper functions
# ============================================================

def make_circle_extrusion(cx, cy, z0, radius, height):
    """Create a vertical cylindrical solid from a circular face."""
    wire = cq.Wire.makeCircle(
        radius,
        center=cq.Vector(cx, cy, z0),
        normal=cq.Vector(0, 0, 1),
    )
    face = cq.Face.makeFromWires(wire)
    solid = cq.Workplane("XY").newObject([face]).extrude(height).val()
    return solid


def edge_center(edge):
    c = edge.Center()
    return c.x, c.y, c.z


def near(a, b, eps=1.0e-5):
    return abs(a - b) <= eps


def edge_radius(edge):
    try:
        return edge.radius()
    except Exception:
        return None


class FinalFilletSelector(cq.Selector):
    """
    Selects the remaining sharp external edges to receive the 0.999 mm fillet.

    Excludes:
    - the original four R5 plate corner-round arc edges
    - tangent boundary edges belonging only to the R5 corner rounds
    - non-sharp internal seam edges
    """

    def filter(self, objectList):
        selected = []

        for edge in objectList:
            bb = edge.BoundingBox()
            cx, cy, cz = edge_center(edge)
            r = edge_radius(edge)
            gtype = edge.geomType()

            z_is_base_bottom = near(cz, z_base_bottom)
            z_is_base_top = near(cz, z_base_top)
            z_is_boss_top = near(cz, z_boss_top)

            horizontal_edge = bb.zlen < 1.0e-5
            vertical_edge = bb.xlen < 1.0e-5 and bb.ylen < 1.0e-5 and bb.zlen > 1.0e-5

            include = False
            exclude = False

            # ------------------------------------------------
            # Exclude original R5 plate corner-round arc edges
            # at top and bottom of the base.
            # ------------------------------------------------
            if horizontal_edge and (z_is_base_bottom or z_is_base_top):
                for qx, qy in corner_hole_centres:
                    if (
                        abs(cx - qx) <= plate_corner_radius + 0.2
                        and abs(cy - qy) <= plate_corner_radius + 0.2
                        and r is not None
                        and near(r, plate_corner_radius, 1.0e-4)
                    ):
                        exclude = True

            # ------------------------------------------------
            # Exclude vertical tangent edges generated by the R5
            # outer plate corner rounds.
            # ------------------------------------------------
            tangent_points = [
                (x_min + plate_corner_radius, y_min),
                (x_min, y_min + plate_corner_radius),
                (x_max - plate_corner_radius, y_min),
                (x_max, y_min + plate_corner_radius),
                (x_max - plate_corner_radius, y_max),
                (x_max, y_max - plate_corner_radius),
                (x_min + plate_corner_radius, y_max),
                (x_min, y_max - plate_corner_radius),
            ]

            if vertical_edge:
                for tx, ty in tangent_points:
                    if near(cx, tx, 1.0e-5) and near(cy, ty, 1.0e-5):
                        exclude = True

            # ------------------------------------------------
            # Base top/bottom straight perimeter edges,
            # excluding the R5 corner arcs above.
            # ------------------------------------------------
            if horizontal_edge and (z_is_base_bottom or z_is_base_top):
                # Long outer edges
                if near(cy, y_min) or near(cy, y_max):
                    if bb.xmin >= x_min + plate_corner_radius - 1.0e-5 and bb.xmax <= x_max - plate_corner_radius + 1.0e-5:
                        include = True

                # Short outer edges, naturally split by the side cutout on +X
                if near(cx, x_min) or near(cx, x_max):
                    if bb.ymin >= y_min + plate_corner_radius - 1.0e-5 and bb.ymax <= y_max - plate_corner_radius + 1.0e-5:
                        include = True

                # Rectangular cutout top/bottom U-profile edges
                if (
                    cutout_x0 - 1.0e-5 <= cx <= cutout_x1 + 1.0e-5
                    and cutout_y0 - 1.0e-5 <= cy <= cutout_y1 + 1.0e-5
                ):
                    include = True

            # ------------------------------------------------
            # Vertical sharp edges of the rectangular cutout.
            # ------------------------------------------------
            if vertical_edge:
                cutout_vertical_points = [
                    (cutout_x0, cutout_y0),
                    (cutout_x0, cutout_y1),
                    (cutout_x1, cutout_y0),
                    (cutout_x1, cutout_y1),
                ]

                for px, py in cutout_vertical_points:
                    if near(cx, px, 1.0e-5) and near(cy, py, 1.0e-5):
                        include = True

            # ------------------------------------------------
            # Corner-hole opening edges.
            # ------------------------------------------------
            if r is not None and near(r, corner_hole_radius, 1.0e-4):
                for hx, hy in corner_hole_centres:
                    if near(cx, hx, 1.0e-5) and near(cy, hy, 1.0e-5):
                        if z_is_base_bottom or z_is_base_top:
                            include = True

            # ------------------------------------------------
            # Central through-hole opening edges only:
            # bottom of base and top of cylinder.
            # The non-sharp internal seam at Z=10 is excluded.
            # ------------------------------------------------
            if r is not None and near(r, central_hole_radius, 1.0e-4):
                if near(cx, boss_cx, 1.0e-5) and near(cy, boss_cy, 1.0e-5):
                    if z_is_base_bottom or z_is_boss_top:
                        include = True

            # ------------------------------------------------
            # Boss external circular sharp edges:
            # base intersection and top outside edge.
            # ------------------------------------------------
            if r is not None and near(r, boss_radius, 1.0e-4):
                if near(cx, boss_cx, 1.0e-5) and near(cy, boss_cy, 1.0e-5):
                    if z_is_base_top or z_is_boss_top:
                        include = True

            if include and not exclude:
                selected.append(edge)

        return selected


# ============================================================
# Operation 1: base plate rectangular face extrusion
# ============================================================

base_pts = [
    (x_min, y_min),
    (x_max, y_min),
    (x_max, y_max),
    (x_min, y_max),
]

base_wire = cq.Workplane("XY").polyline(base_pts).close().objects[0]
base_face = cq.Face.makeFromWires(base_wire)
base_solid = cq.Workplane("XY").newObject([base_face]).extrude(base_thickness).val()

part = cq.Workplane("XY").newObject([base_solid])

# ============================================================
# Operation 2: create four R5 outer plate corner rounds
# ============================================================

part = part.edges("|Z").fillet(plate_corner_radius)

# ============================================================
# Operation 3: subtract four Ø6 corner through-holes
# ============================================================

for hx, hy in corner_hole_centres:
    hole_tool = make_circle_extrusion(
        cx=hx,
        cy=hy,
        z0=z_base_bottom,
        radius=corner_hole_radius,
        height=base_thickness,
    )
    part = part.cut(hole_tool)

# ============================================================
# Operation 4: add vertical Ø20 boss cylinder on top face
# ============================================================

boss_solid = make_circle_extrusion(
    cx=boss_cx,
    cy=boss_cy,
    z0=z_base_top,
    radius=boss_radius,
    height=boss_height,
)

part = part.union(boss_solid)

# ============================================================
# Operation 5: subtract concentric Ø15 through-hole
# through boss and base
# ============================================================

central_hole_tool = make_circle_extrusion(
    cx=boss_cx,
    cy=boss_cy,
    z0=z_base_bottom,
    radius=central_hole_radius,
    height=z_boss_top - z_base_bottom,
)

part = part.cut(central_hole_tool)

# ============================================================
# Operation 6: rectangular right-edge cutout
# 20 mm side parallel to short edge, full base thickness
# ============================================================

cutout_pts = [
    (cutout_x0, cutout_y0),
    (cutout_x1, cutout_y0),
    (cutout_x1, cutout_y1),
    (cutout_x0, cutout_y1),
]

cutout_wire = cq.Workplane("XY").polyline(cutout_pts).close().objects[0]
cutout_face = cq.Face.makeFromWires(cutout_wire)
cutout_solid = cq.Workplane("XY").newObject([cutout_face]).extrude(base_thickness).val()

part = part.cut(cutout_solid)

# ============================================================
# Operation 7: final 0.999 mm fillet on remaining sharp edges
# excluding the original R5 plate corner rounds
# ============================================================

part = part.edges(FinalFilletSelector()).fillet(final_fillet_radius)

# Final model
result = part


cq.exporters.export(result, "S1-R05A.stl")
