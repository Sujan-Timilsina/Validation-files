import cadquery as cq

# -------------------------------------------------------------------------
# CUSTOM SELECTOR FOR DETERMINISTIC EDGE SELECTION
# -------------------------------------------------------------------------
class SharpExternalEdges(cq.Selector):
    """
    Selects all sharp external edges to be filleted (horizontal edges, circular edges, 
    and the vertical edges of the rectangular cutout).
    Explicitly excludes the vertical tangent seams generated by the R5 corner rounds 
    and the vertical tangent seams of the main cylinders.
    """
    def filter(self, objectList):
        res = []
        for o in objectList:
            # Circles and ellipses (hole edges, cylinder caps, and R5 top/bottom arcs)
            if o.geomType() in ['CIRCLE', 'ELLIPSE']:
                res.append(o)
            
            # Straight lines (plate borders and vertical lines)
            elif o.geomType() == 'LINE':
                p1, p2 = o.startPoint(), o.endPoint()
                
                # Check if the line is vertical (parallel to Z axis)
                is_vertical = abs(p1.x - p2.x) < 1e-4 and abs(p1.y - p2.y) < 1e-4
                
                if is_vertical:
                    c = o.Center()
                    # Only append vertical edges if they belong to the rectangular cutout.
                    # The cutout exists between X=80 to 90, and Y=15 to 35.
                    if 79 <= c.x <= 91 and 14 <= c.y <= 36:
                        res.append(o)
                else:
                    # All horizontal straight edges are sharp edges
                    res.append(o)
                    
        return res

def create_part():
    # -------------------------------------------------------------------------
    # 1. BASE PLATE & RECTANGULAR CUTOUT
    # Origin (0,0,0) at bottom-left corner.
    # Cutout is 20x10 at the center of the right short edge (X=80 to 90, Y=15 to 35).
    # -------------------------------------------------------------------------
    pts = [
        (0, 0),     # Bottom-left
        (90, 0),    # Bottom-right
        (90, 15),   # Cutout start
        (80, 15),   # Cutout inner corner
        (80, 35),   # Cutout inner corner
        (90, 35),   # Cutout end
        (90, 50),   # Top-right
        (0, 50)     # Top-left
    ]
    
    # Mandatory Profile Construction Protocol
    wire_base = cq.Workplane("XY").polyline(pts).close().objects[0]
    face_base = cq.Face.makeFromWires(wire_base)
    base_solid = cq.Workplane("XY").newObject([face_base]).extrude(10)

    # -------------------------------------------------------------------------
    # 2. OUTER CORNER ROUNDS (R5)
    # Target the 4 outer vertical edges at the bounding box corners.
    # -------------------------------------------------------------------------
    base_solid = base_solid.edges("|Z").edges(
        cq.selectors.NearestToPointSelector((0, 0, 5)) |
        cq.selectors.NearestToPointSelector((90, 0, 5)) |
        cq.selectors.NearestToPointSelector((90, 50, 5)) |
        cq.selectors.NearestToPointSelector((0, 50, 5))
    ).fillet(5)

    # -------------------------------------------------------------------------
    # 3. CORNER THROUGH-HOLES (D6 -> R3)
    # Concentric to the R5 fillets.
    # -------------------------------------------------------------------------
    hole_pts = [(5, 5), (85, 5), (85, 45), (5, 45)]
    for p in hole_pts:
        wire_h = cq.Workplane("XY").center(p[0], p[1]).circle(3).objects[0]
        face_h = cq.Face.makeFromWires(wire_h)
        cyl_h = cq.Workplane("XY").newObject([face_h]).extrude(10)
        base_solid = base_solid.cut(cyl_h)

    part = base_solid

    # -------------------------------------------------------------------------
    # 4. VERTICAL CYLINDER (D20 -> R10)
    # Height = 50, originating from top face of plate (Z=10 to Z=60).
    # -------------------------------------------------------------------------
    wire_cyl = cq.Workplane("XY").center(30, 20).workplane(offset=10).circle(10).objects[0]
    face_cyl = cq.Face.makeFromWires(wire_cyl)
    top_cyl = cq.Workplane("XY").workplane(offset=10).newObject([face_cyl]).extrude(50)
    
    # Watertight attachment
    part = part.union(top_cyl)

    # -------------------------------------------------------------------------
    # 5. CYLINDER THROUGH-HOLE (D15 -> R7.5)
    # Passes through cylinder and plate (Z=0 to Z=60).
    # -------------------------------------------------------------------------
    wire_ch = cq.Workplane("XY").center(30, 20).circle(7.5).objects[0]
    face_ch = cq.Face.makeFromWires(wire_ch)
    center_hole = cq.Workplane("XY").newObject([face_ch]).extrude(60)
    
    part = part.cut(center_hole)

    # -------------------------------------------------------------------------
    # 6. GLOBAL FILLET
    # Apply to all sharp external edges using the custom selector.
    # Using CHAMFER TOLERANCE RULE: 1 mm - 0.001 mm = 0.999 mm
    # -------------------------------------------------------------------------
    part = part.edges(SharpExternalEdges()).fillet(0.999)

    return part

# Execute generation
result = create_part()

cq.exporters.export(result, "S1-R10C.stl")
